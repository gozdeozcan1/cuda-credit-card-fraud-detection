cmake_minimum_required(VERSION 3.24)
project(CudaCreditCardFraud LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_OPENMP "Build OpenMP executables" ON)
option(BUILD_CUDA "Build CUDA executables" OFF)
if(BUILD_OPENMP)
  find_package(OpenMP REQUIRED)
endif()
if(BUILD_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
endif()
foreach(model IN ITEMS LR SVM RF)
  string(TOLOWER "${model}" lower_model)
  set(serial_target "fraud_${lower_model}_serial")
  add_executable(${serial_target} src/main.cpp src/Common.cpp src/Serial${model}.cpp)
  target_compile_definitions(${serial_target} PRIVATE FRAUD_MODEL_${model})
  if(BUILD_OPENMP)
    set(omp_target "fraud_${lower_model}_openmp")
    add_executable(${omp_target} src/main.cpp src/Common.cpp src/${model}OpenMP.cpp)
    target_compile_definitions(${omp_target} PRIVATE FRAUD_MODEL_${model} FRAUD_OPENMP)
    target_link_libraries(${omp_target} PRIVATE OpenMP::OpenMP_CXX)
  endif()
  if(BUILD_CUDA)
    set(cuda_target "fraud_${lower_model}_cuda")
    add_executable(${cuda_target} src/main.cpp src/Common.cpp src/${model}CUDA.cu)
    target_compile_definitions(${cuda_target} PRIVATE FRAUD_MODEL_${model} FRAUD_CUDA)
    set_target_properties(${cuda_target} PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    target_link_libraries(${cuda_target} PRIVATE CUDA::cudart)
  endif()
endforeach()
